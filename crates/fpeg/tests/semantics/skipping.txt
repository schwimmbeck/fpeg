=== the skip expression runs before every token and at the end
start: "a" "b"
%ignore " "+
--- " a  b  "
(start ~"a" ~"b")
--- spans " a  b  "
(start@1..5 ~"a"@1..2 ~"b"@4..5)

=== several ignore directives form one choice
start: "a" "b"
%ignore " "+
%ignore "#" [^\n]* "\n"
--- "a # note\n b"
(start ~"a" ~"b")

=== a terminal skips nothing inside
start: WORD WORD
WORD: [a-z]+
%ignore " "+
--- "ab cd"
(start WORD"ab" WORD"cd")
--- "a b c"
error at 4: expected end of input

=== a lexical rule produces a node but skips nothing between its elements
start: pair
@lexical
pair: KEY "=" KEY
KEY: [a-z]+
%ignore " "+
--- "  a=b"
(start (pair KEY"a" ~"=" KEY"b"))
--- "a = b"
error at 1: expected "="

=== a lexical rule called from a syntactic rule skips before itself
start: quoted+
@lexical
quoted: "\"" TEXT? (hole TEXT?)* "\""
hole: "{" NAME "}"
TEXT: [^"{]+
NAME: [a-z]+
%ignore " "+
--- " \"a { x } b\" \"c\""
(start (quoted ~"\"" TEXT"a " (hole ~"{" NAME"x" ~"}") TEXT" b" ~"\"") (quoted ~"\"" TEXT"c" ~"\""))

=== a grammar without ignore skips nothing
start: "a" "b"
--- "a b"
error at 1: expected "b"

=== the skip expression may use a template
start: "a" "b"
%ignore SPACES<" ">
SPACES<x>: x+
--- "a   b"
(start ~"a" ~"b")

=== what follows a rule that skips and matches nothing starts after the skip
start: m
@lexical
m: "x" (s "b")? "z"
s: T
T: ()
%ignore " "+
--- "x bz"
(start (m ~"x" (s T"") ~"b" ~"z"))
--- "x z"
error at 2: expected "b"

=== a skip under a flag in a part that matches nothing moves what follows the part
%flag f
start: "c" | %on(f, T) "b" | "d"
T: ()
%ignore " "+ | %if(f) "\n"
--- "\nb"
(start T"" ~"b")
--- "\nd"
error at 1: expected "b"

=== a left recursion goes on after a rule that skips and matches nothing
start: m
@lexical
m: "y" e
@lexical
e: e s "b" | "x"
s: T
T: ()
%ignore " "+
--- "yx b"
(start (m ~"y" (e (e ~"x") (s T"") ~"b")))
--- "yx c"
error at 3: expected "b", end of input
