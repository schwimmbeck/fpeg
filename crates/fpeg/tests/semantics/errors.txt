=== an error label stops the parse with its message
start: "a" ("b" ^ "b is missing")
--- "ac"
error at 1: b is missing
--- "ab"
(start ~"a" ~"b")

=== a named error label stops the parse with its name
start: "a" ("b" ^ missing_b)
--- "a"
error at 1: missing_b: expected "b"

=== a labeled error is not caught by a choice
start: "a" ("b" ^ "need b") | "a" "c"
--- "ac"
error at 1: need b

=== a cut inside a labeled expression does not cancel the label
start: ("if" ~ !"z" "(") ^ "bad if" | "if" "z"
--- "ifz"
error at 0: bad if

=== a labeled error gives way to the farthest failure, where the parse got further than the label
start: ("a" "b" "c") ^ abc
--- "abx"
error at 2: expected "c"
--- "x"
error at 0: abc: expected "a"

=== a cut inside a labeled expression that matches commits the choice around it
start: ("a" ~ "b") ^ "bad" "x" | "a" "b" "y"
--- "aby"
error at 2: expected "x"

=== a labeled error points past ignored text
start: "a" ("b" ^ "b is missing")
%ignore " "+
--- "a   c"
error at 4: b is missing

=== a labeled error inside a lookahead is a failure of the lookahead
start: !("a" ^ "never") "b"
--- "b"
(start ~"b")

=== an error alternative never matches, and explains a failed parse
start: "a" | %error("no a here")
--- "a"
(start ~"a")
--- "b"
error at 0: no a here

=== line breaks inside the parentheses of the forms that start with % are skipped
%flag f
start: %on(f, %if(
    f
) "a") | %error(
    "no a here"
)
--- "a"
(start ~"a")
--- "b"
error at 0: no a here

=== the first error alternative that matches explains the error, over what it matched
start: "a" ("b" %error("not b") | "b" "c" %error("not bc") | "d")
--- "abc"
error at 1..2: not b

=== the attempt that comes first in the text explains the error, whatever was tried first
start: "a" ("b" "c" ("d" | %error("far")) | %error("near"))
--- "abcx"
error at 1: near

=== an error alternative's error starts after the ignored text, as a label's does
start: assignment+
assignment: NAME "=" NUMBER ";" | NAME "==" %error("= assigns, == compares")
NAME: [a-z]+
NUMBER: [0-9]+
%ignore " "+
--- "a = 1; x == 2;"
error at 7..11: = assigns, == compares

=== an error alternative tried in several contexts starts after the farthest ignored text of them
%ignore "\n" [ \t]* %indent(())
start: %indent(start) | () %error("e2")
--- "\n\t"
error at 2: e2

=== whichever context sorts or comes first, the farthest skip of them starts the error
%flag f
%ignore %if(f) " "
start: %on(f, start) | () %error("e2")
--- " "
error at 1: e2

=== attempts at one position are tested in the order their alternatives stand in the grammar
pair<x>: x x | "b" %error("first")
start: pair<"a"> | "b" %error("second")
--- "b"
error at 0..1: first

=== an attempt is tested in the context it was made in, with its captures
start: %def(q, "'") ("x" %ref(q) | "y" %ref(q) %error("y is no x"))
--- "'y'"
error at 1..3: y is no x

=== an error alternative inside a lookahead explains nothing
start: !("a" %error("never shown")) "b"
--- "a"
error at 0: expected "b"

=== an attempt counts where a rule matched before inside a lookahead runs again outside it
start: !"b" "a"
%ignore " " | C
@memo
C: "#" "x"+ | "#" %error("a comment needs text")
--- "#"
error at 0..1: a comment needs text

=== a label inside an error alternative fails it
start: "a" (("b" ^ "never") %error("not b") | "c" | "x" %error("x is no c"))
--- "ax"
error at 1..2: x is no c

=== an error alternative in the skip expression explains the error after it
start: "a" "b"
%ignore " "+ | "/*" (!"*/" .)* "*/" | "/*" (!"*/" .)* %error("the comment does not end")
--- "a /* b"
error at 2..6: the comment does not end

=== an error alternative in a terminal explains the error at its token
start: STRING+
STRING: "'" [a-z]* "'" | "'" [a-z]* %error("the string does not end")
--- "'ab"
error at 0..3: the string does not end

=== the end of the input says it was expected where text is left
start: "a" !. | "a" "b" "c"
%ignore " "+
--- "a b x"
error at 4: expected "c"
--- "a x"
error at 2: expected "b", end of input
--- "a "
(start ~"a")

=== the end of the input records what it expected also where the next byte rules out what follows
start: T
T: !. "x" | "y"
--- "z"
error at 0: expected T, end of input

=== the expected set lists everything that failed at the farthest position
start: "a" ("b" | "c" | D)
D "digit": [0-9]
--- "ax"
error at 1: expected "b", "c", digit

=== a reserved terminal never matches a keyword
start: NAME*
@reserved
NAME: [a-z]+
other: "if"
%ignore " "+
--- "x if"
error at 2: expected NAME, end of input

=== a labeled error inside a terminal inside a lookahead fails the lookahead
start: !T "a" "c" | "a" "b"
T: "a" ("x" ^ "need x")
--- "ab"
(start ~"a" ~"b")

=== a soft keyword stays a valid name
start: NAME*
@reserved
NAME: [a-z]+
other: "match"
%keywords soft "match"
%ignore " "+
--- "match x"
(start NAME"match" NAME"x")

=== the expected set holds the tokens of every alternative that failed at the position
start: "x" "1" | "y" "2" | "z"
--- "w"
error at 0: expected "x", "y", "z"

=== alternatives after one that matches nothing are never tried and expect nothing
start: ("a" | () | "b") "c" | "d"
--- "e"
error at 0: expected "a", "c", "d"

=== a token that matches nothing expects nothing
start: A "x" | "z"
A: "a"*
--- "y"
error at 0: expected "x", "z"

=== a token whose lookahead fails is expected, though it could match nothing
start: T "x" | "z"
T: !"y"
--- "y"
error at 0: expected "z", T

=== a repetition of no steps never tries its expression
start: "a"{0} "b" | "c"
--- "d"
error at 0: expected "b", "c"

=== a left-recursive call where its rule started gives the seed and tries no token
start: start | (start "b")?
--- "x"
error at 0: expected end of input

=== a label throws where its expression fails at the first byte
start: A | "b"
A: ("x" ^ "need x") "y"
--- "b"
error at 0: need x
