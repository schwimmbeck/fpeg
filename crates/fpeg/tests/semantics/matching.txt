=== a sequence of literals produces a node of hidden tokens
start: "a" "b"
--- "ab"
(start ~"a" ~"b")
--- "a"
error at 1: expected "b"
--- "abc"
error at 2: expected end of input

=== ordered choice takes the first alternative that matches
start: "a" | "a" "b"
--- "a"
(start ~"a")
--- "ab"
error at 1: expected end of input

=== repetition is greedy and never gives back
start: "a"* "a"
--- "aaa"
error at 3: expected "a"

=== bounded repetition takes between the bounds
start: "a"{2,3}
--- "a"
error at 1: expected "a"
--- "aa"
(start ~"a" ~"a")
--- "aaaa"
error at 3: expected end of input

=== an exact count takes exactly that many
start: "a"{2} "b"
--- "aab"
(start ~"a" ~"a" ~"b")
--- "aaab"
error at 2: expected "b"

=== separated repetition takes no trailing separator
start: "[" ("x" ** ",") "]"
--- "[]"
(start ~"[" ~"]")
--- "[x,x]"
(start ~"[" ~"x" ~"," ~"x" ~"]")
--- "[x,]"
error at 3: expected "x"

=== one or more separated items need at least one
start: "x" ++ ","
--- ""
error at 0: expected "x"
--- "x,x"
(start ~"x" ~"," ~"x")

=== lookahead consumes nothing and records nothing
start: !"a" . | "a" "b"
--- "b"
(start ~"b")
--- "ac"
error at 1: expected "b"

=== positive lookahead tests without consuming
start: &"ab" "a" .
--- "ab"
(start ~"a" ~"b")
--- "ac"
error at 0: no match

=== a cut commits the choice
start: "a" ~ "b" | "a" "c"
--- "ab"
(start ~"a" ~"b")
--- "ac"
error at 1: expected "b"

=== a cut inside an optional makes it fail instead of succeed empty
start: ("a" ~ "b")? "a" "c"
--- "ac"
error at 1: expected "b"

=== without a cut the optional succeeds empty
start: ("a" "b")? "a" "c"
--- "ac"
(start ~"a" ~"c")

=== a cut in the last alternative of a group commits the group alone
start: ("a" | "b" ~ "c") | "b" "d"
--- "bd"
(start ~"b" ~"d")
--- "bc"
(start ~"b" ~"c")

=== a cut that a group has taken does not commit the choice around it
start: ("a" | "b" ~) "c" | "b" "d"
--- "bd"
(start ~"b" ~"d")

=== a cut inside an expression repeated once ends with it
start: ("a" ~){1} "b" | "a" "c"
--- "ac"
(start ~"a" ~"c")

=== a cut inside a repetition makes the whole repetition fail
start: ("a" ~ "b")* "a" "c" | "a" "b" "a" "d"
--- "abad"
(start ~"a" ~"b" ~"a" ~"d")
--- "abac"
error at 3: expected "b", "d"

=== a cut in a step that matched is spent
start: ("a" ~ "b")* "d"
--- "abd"
(start ~"a" ~"b" ~"d")

=== a cut in a counted step that matched is spent
start: ("a" ~ "b"){1,3} "d"
--- "abd"
(start ~"a" ~"b" ~"d")

=== a cut inside a counted repetition makes it fail though it has steps enough
start: ("a" ~ "b"){1,2} "a" | "a" "b" "a" "c"
--- "aba"
error at 3: expected "b", "c"

=== a cut does not reach past the lookahead it stands in
start: &("a" ~ "b") "a" "b" | "a" "c"
--- "ac"
(start ~"a" ~"c")

=== a cut does not reach the rule that calls its rule
start: pair | "a" "c"
pair: "a" ~ "b"
--- "ac"
(start ~"a" ~"c")

=== a cut in the item of a separated list commits the list
start: ("a" ~ "b") ** "," "c" | "a" "c"
--- "ac"
(start ~"a" ~"c")
--- "ab,ac"
error at 4: expected "b"

=== a cut in the ignored text ends the skipping where it fails
start: "a" "b"
%ignore " " ~ " "
--- "a  b"
(start ~"a" ~"b")
--- "a b"
error at 1: expected "b"

=== a repetition step that matches empty is the last
start: ("a"?)* "b"
--- "b"
(start ~"b")
--- "aab"
(start ~"a" ~"a" ~"b")

=== a repetition of a repetition is not a separated repetition
start: ("a"+)+ ("b"*)* "c"
--- "aac"
(start ~"a" ~"a" ~"c")

=== a case-insensitive literal ignores ASCII case
start: "select"i
--- "SeLeCt"
(start ~"SeLeCt")

=== a class matches one code point and spans are bytes
start: \p{XID_Start} [^a]
--- spans "äö"
(start@0..4 ~"ä"@0..2 ~"ö"@2..4)

=== each property matches the code points that have it
start: \p{Alphabetic} \p{White_Space} \p{Uppercase} \p{Lowercase} \P{Alphabetic}
--- "中\tÄß1"
(start ~"中" ~"\t" ~"Ä" ~"ß" ~"1")
--- "中\tßß1"
error at 4: expected \p{Uppercase}

=== a literal takes every escape of the table
start: "\\" "\"" '\'' "\n" "\r" "\t" "\0" "\x41" "\u{e9}"
--- "\\\"'\n\r\t\0Aé"
(start ~"\\" ~"\"" ~"'" ~"\n" ~"\r" ~"\t" ~"\0" ~"A" ~"é")

=== any matches one code point and fails at the end
start: . .
--- "€x"
(start ~"€" ~"x")
--- "€"
error at 3: expected .

=== a literal ending in a word character needs a boundary
start: "if" NAME
NAME: [a-z]+
%word [A-Za-z0-9_]
%ignore " "+
--- "if x"
(start ~"if" NAME"x")
--- "iffy"
error at 0: expected "if"

=== the word characters can be changed
start: "if" NAME
NAME: [a-z]+
%word [a-z]
%ignore " "+
--- "if_x"
error at 2: expected NAME
--- "if x"
(start ~"if" NAME"x")
